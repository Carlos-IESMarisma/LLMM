const xhttp = new XMLHttpRequest();
xhttp.onload = function() {
  const books = this.responseXML.getElementsByTagName("book");
  let html = "<table border='1'><tr><th>Título</th><th>Autor</th><th>Precio</th><th>Año</th></tr>";
  for(let i = 0; i < books.length; i++) {
    let titulo = books[i].getElementsByTagName("title")[0].childNodes[0].nodeValue;
    let autor  = books[i].getElementsByTagName("author")[0].childNodes[0].nodeValue;
    let precio = books[i].getElementsByTagName("price")[0].childNodes[0].nodeValue;
    let anio   = books[i].getElementsByTagName("year")[0].childNodes[0].nodeValue;
    html += "<tr><td>"+titulo+"</td><td>"+autor+"</td><td>"+precio+"€</td><td>"+anio+"</td></tr>";
  }
  html += "</table>";
  document.getElementById("demo").innerHTML = html;
};
xhttp.open("GET", "https://www.w3schools.com/xml/books.xml");
xhttp.send();
</script></body></html>
