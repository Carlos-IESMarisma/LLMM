1. var titulos = "";
var n = document.getElementsByTagName("TITLE").length;
for(var i = 0; i < n; i++)
  titulos += document.getElementsByTagName("TITLE")[i].childNodes[0].nodeValue + " <br> ";
document.getElementById("demo").innerHTML = titulos;
